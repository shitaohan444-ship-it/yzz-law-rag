import hashlib
import json
import os
from pathlib import Path

from app.core.config import settings


def resolve_model_path(reference: str):
    direct = Path(reference).expanduser()

    if direct.exists():
        return direct.resolve(), "DIRECT_PATH"

    hf_home = os.environ.get("HF_HOME")
    hf_cache = os.environ.get("HF_HUB_CACHE")

    if hf_cache:
        cache_root = Path(hf_cache)
    elif hf_home:
        cache_root = Path(hf_home) / "hub"
    else:
        cache_root = Path.home() / ".cache" / "huggingface" / "hub"

    repo_directory = (
        "models--" + reference.replace("/", "--")
    )

    repo_root = cache_root / repo_directory
    snapshots_root = repo_root / "snapshots"

    if not snapshots_root.exists():
        return None, "NOT_FOUND"

    main_ref = repo_root / "refs" / "main"

    if main_ref.exists():
        commit = main_ref.read_text(encoding="utf-8").strip()
        snapshot = snapshots_root / commit

        if snapshot.exists():
            return snapshot.resolve(), "HF_CACHE_MAIN"

    snapshots = sorted(
        (path for path in snapshots_root.iterdir() if path.is_dir()),
        key=lambda path: path.stat().st_mtime,
        reverse=True,
    )

    if snapshots:
        return snapshots[0].resolve(), "HF_CACHE_LATEST"

    return None, "NOT_FOUND"


def sha256_file(path: Path):
    digest = hashlib.sha256()

    with path.open("rb") as file:
        for block in iter(lambda: file.read(1024 * 1024), b""):
            digest.update(block)

    return digest.hexdigest()
