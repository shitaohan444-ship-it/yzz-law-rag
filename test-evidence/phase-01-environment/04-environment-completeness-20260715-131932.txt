PS D:\law-rag-project> function Read-DotEnv {
    param([string]$Path)

    $Map = @{}

    if (-not (Test-Path -LiteralPath $Path)) {
        return $Map
    }

    foreach ($RawLine in Get-Content -LiteralPath $Path) {
        $Line = $RawLine.Trim()

        if (-not $Line -or $Line.StartsWith("#")) {
            continue
        }

        if ($Line.StartsWith("export ")) {
            $Line = $Line.Substring(7).Trim()
        }

        $Index = $Line.IndexOf("=")

        if ($Index -le 0) {
            continue
        }

        $Key = $Line.Substring(0, $Index).Trim()
        $Value = $Line.Substring($Index + 1).Trim()

        if ($Key -match "^[A-Za-z_][A-Za-z0-9_]*$") {
            $Map[$Key] = $Value
        }
    }

    return $Map
}
PS D:\law-rag-project> function Get-ValueStatus {
    param([string]$Value)

    $CleanValue = $Value.Trim().Trim('"').Trim("'")

    if ([string]::IsNullOrWhiteSpace($CleanValue)) {
        return "EMPTY"
    }

    $PlaceholderPattern =
        "(?i)(change[_ -]?me|replace[_ -]?me|" +
        "your[_ -]?(key|token|password|secret)|" +
        "placeholder|请填写|你的.*(密钥|密码)|xxx+)"

    if ($CleanValue -match $PlaceholderPattern) {
        return "PLACEHOLDER"
    }

    return "SET"
}
