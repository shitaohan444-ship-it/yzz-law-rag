from importlib.metadata import PackageNotFoundError, version
from pathlib import Path
from pip._vendor.packaging.requirements import Requirement

path = Path(r".\law-rag-kb\requirements.txt")
failed = []
checked = 0
skipped = 0

for line_no, raw in enumerate(path.read_text(encoding="utf-8-sig").splitlines(), 1):
    line = raw.strip()

    if not line or line.startswith("#"):
        continue

    if line.startswith(("-r ", "--", "-e ")):
        print(f"[SKIP] line={line_no} directive")
        skipped += 1
        continue

    line = line.split(" #", 1)[0].strip()

    try:
        req = Requirement(line)
    except Exception as exc:
        print(f"[PARSE-FAIL] line={line_no} type={type(exc).__name__}")
        failed.append((line_no, "parse"))
        continue

    if req.marker and not req.marker.evaluate():
        print(f"[SKIP] {req.name} marker_not_applicable")
        skipped += 1
        continue

    checked += 1

    try:
        installed = version(req.name)
    except PackageNotFoundError:
        print(f"[MISSING] {req.name}")
        failed.append((req.name, "missing"))
        continue

    compatible = not req.specifier or req.specifier.contains(
        installed,
        prereleases=True
    )

    status = "OK" if compatible else "VERSION-MISMATCH"
    expected = str(req.specifier) if req.specifier else "(any)"
    print(
        f"[{status}] {req.name} "
        f"installed={installed} required={expected}"
    )

    if not compatible:
        failed.append((req.name, "version"))

print(
    f"SUMMARY checked={checked} skipped={skipped} "
    f"failed={len(failed)}"
)

raise SystemExit(1 if failed else 0)
